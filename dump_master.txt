def dataset_maps(image_dir, mask_dir, input_root):
    def maps(path):
        result = {}
        for p in Path(path).iterdir():
            if p.is_file() and p.suffix.lower() in [
                ".jpg",
                ".jpeg",
                ".png",
                ".bmp",
                ".tif",
                ".tiff",
            ]:
                if p.stem in result:
                    raise ValueError("Duplicate ID: " + p.stem)
                result[p.stem] = str(p)
        return result

    if bool(image_dir) != bool(mask_dir):
        raise ValueError("Set both images and masks, or leave both None")
    candidates = (
        [(Path(image_dir), Path(mask_dir))]
        if image_dir
        else [
            (p, p.parent / "masks")
            for p in Path(input_root).rglob("images")
            if "kvasir" in str(p).lower() and (p.parent / "masks").is_dir()
        ]
    )
    good = []
    for a, b in candidates:
        im, ma = maps(a), maps(b)
        if len(set(im) & set(ma)) == 1000 and set(im) == set(ma):
            good.append((im, ma))
    if len(good) != 1:
        raise ValueError(
            "Set PATHS images/masks: expected exactly one paired Kvasir1000 dataset"
        )
    return good[0]

def make_split(images, masks, seed):
    ids = np.asarray(sorted(set(images) & set(masks)))
    if len(ids) != 1000 or set(images) != set(masks):
        raise ValueError("Expected 1000 correctly paired IDs")
    order = np.random.default_rng(seed).permutation(len(ids))
    train = ids[order[:900]].tolist()
    test = ids[order[900:]].tolist()
    assert not set(train) & set(test) and len(set(train + test)) == 1000
    return {
        "dataset": "Kvasir-SEG",
        "seed": int(seed),
        "train_count": 900,
        "test_count": 100,
        "method": "sorted matched IDs -> default_rng(seed).permutation -> first900/rest100",
        "train": train,
        "test": test,
    }

def tensor_pair(record, size, seed=None, synthetic=None):
    with Image.open(synthetic or record["image"]) as handle:
        image = handle.convert("RGB").resize((size, size), Image.Resampling.BILINEAR)
    with Image.open(record["mask"]) as handle:
        mask = handle.convert("L").resize((size, size), Image.Resampling.NEAREST)
    if seed is not None:
        rng = random.Random(int(seed))
        if rng.random() < 0.5:
            image = image.transpose(Image.Transpose.FLIP_LEFT_RIGHT)
            mask = mask.transpose(Image.Transpose.FLIP_LEFT_RIGHT)
        angle = rng.uniform(-45, 45)
        image = image.rotate(angle, Image.Resampling.BILINEAR)
        mask = mask.rotate(angle, Image.Resampling.NEAREST)
    x = torch.from_numpy(np.asarray(image).copy().transpose(2, 0, 1)).float() / 255
    y = torch.from_numpy((np.asarray(mask) > 127).astype("float32").copy())[None]
    return x, y


def stream_ids(ids, positions, seed):
    cache = {}
    chosen = []
    for position in positions:
        epoch, index = divmod(int(position), len(ids))
        if epoch not in cache:
            cache[epoch] = np.random.default_rng(seed + epoch).permutation(len(ids))
        chosen.append(ids[int(cache[epoch][index])])
    return chosen

def timestep_embedding(timesteps, dim, max_period=1000):
    """
    Create sinusoidal timestep embeddings.
    :param timesteps: a 1-D Tensor of N indices, one per batch element.
                      These may be fractional.
    :param dim: the dimension of the output.
    :param max_period: controls the minimum frequency of the embeddings.
    :return: an [N x dim] Tensor of positional embeddings.
    """
    half = dim // 2
    freqs = th.exp(
        -math.log(max_period) * th.arange(start=0, end=half, dtype=th.float32) / half
    ).to(device=timesteps.device)
    args = timesteps[:, None].float() * freqs[None]
    embedding = th.cat([th.cos(args), th.sin(args)], dim=-1)
    if dim % 2:
        embedding = th.cat([embedding, th.zeros_like(embedding[:, :1])], dim=-1)
    return embedding


class ResBlock(TimestepBlock):
    """
    A residual block that can optionally change the number of channels.
    :param channels: the number of input channels.
    :param emb_channels: the number of timestep embedding channels.
    :param dropout: the rate of dropout.
    :param out_channels: if specified, the number of out channels.
    :param use_conv: if True and out_channels is specified, use a spatial
        convolution instead of a smaller 1x1 convolution to change the
        channels in the skip connection.
    :param dims: determines if the signal is 1D, 2D, or 3D.
    :param use_checkpoint: if True, use gradient checkpointing on this module.
    :param up: if True, use this block for upsampling.
    :param down: if True, use this block for downsampling.
    """

    def __init__(
        self,
        channels,
        emb_channels,
        dropout,
        out_channels=None,
        use_conv=False,
        use_scale_shift_norm=False,
        dims=2,
        use_checkpoint=False,
        up=False,
        down=False,
    ):
        super().__init__()
        self.channels = channels
        self.emb_channels = emb_channels
        self.dropout = dropout
        self.out_channels = out_channels or channels
        self.use_conv = use_conv
        self.use_checkpoint = use_checkpoint
        self.use_scale_shift_norm = use_scale_shift_norm

        self.in_layers = nn.Sequential(
            normalization(channels),
            nn.SiLU(),
            conv_nd(dims, channels, self.out_channels, 3, padding=1),
        )

        self.updown = up or down

        if up:
            self.h_upd = Upsample(channels, False, dims)
            self.x_upd = Upsample(channels, False, dims)
        elif down:
            self.h_upd = Downsample(channels, False, dims)
            self.x_upd = Downsample(channels, False, dims)
        else:
            self.h_upd = self.x_upd = nn.Identity()

        self.emb_layers = nn.Sequential(
            nn.SiLU(),
            linear(
                emb_channels,
                2 * self.out_channels if use_scale_shift_norm else self.out_channels,
            ),
        )
        self.out_layers = nn.Sequential(
            normalization(self.out_channels),
            nn.SiLU(),
            nn.Dropout(p=dropout),
            zero_module(
                conv_nd(dims, self.out_channels, self.out_channels, 3, padding=1)
            ),
        )

        if self.out_channels == channels:
            self.skip_connection = nn.Identity()
        elif use_conv:
            self.skip_connection = conv_nd(
                dims, channels, self.out_channels, 3, padding=1
            )
        else:
            self.skip_connection = conv_nd(dims, channels, self.out_channels, 1)

    def forward(self, x, emb):
        """
        Apply the block to a Tensor, conditioned on a timestep embedding.
        :param x: an [N x C x ...] Tensor of features.
        :param emb: an [N x emb_channels] Tensor of timestep embeddings.
        :return: an [N x C x ...] Tensor of outputs.
        """
        return checkpoint(
            self._forward, (x, emb), self.parameters(), self.use_checkpoint
        )

    def _forward(self, x, emb):
        if self.updown:
            in_rest, in_conv = self.in_layers[:-1], self.in_layers[-1]
            h = in_rest(x)
            h = self.h_upd(h)
            x = self.x_upd(x)
            h = in_conv(h)
        else:
            h = self.in_layers(x)
        emb_out = self.emb_layers(emb).type(h.dtype)
        while len(emb_out.shape) < len(h.shape):
            emb_out = emb_out[..., None]
        if self.use_scale_shift_norm:
            out_norm, out_rest = self.out_layers[0], self.out_layers[1:]
            scale, shift = th.chunk(emb_out, 2, dim=1)
            h = out_norm(h) * (1 + scale) + shift
            h = out_rest(h)
        else:
            h = h + emb_out
            h = self.out_layers(h)
        return self.skip_connection(x) + h

class UNetModel(nn.Module):
    """
    The full UNet model with attention and timestep embedding.
    :param in_channels: channels in the input Tensor.
    :param model_channels: base channel count for the model.
    :param out_channels: channels in the output Tensor.
    :param num_res_blocks: number of residual blocks per downsample.
    :param attention_resolutions: a collection of downsample rates at which
        attention will take place. May be a set, list, or tuple.
        For example, if this contains 4, then at 4x downsampling, attention
        will be used.
    :param dropout: the dropout probability.
    :param channel_mult: channel multiplier for each level of the UNet.
    :param conv_resample: if True, use learned convolutions for upsampling and
        downsampling.
    :param dims: determines if the signal is 1D, 2D, or 3D.
    :param num_classes: if specified (as an int), then this model will be
        class-conditional with `num_classes` classes.
    :param use_checkpoint: use gradient checkpointing to reduce memory usage.
    :param num_heads: the number of attention heads in each attention layer.
    :param num_heads_channels: if specified, ignore num_heads and instead use
                               a fixed channel width per attention head.
    :param num_heads_upsample: works with num_heads to set a different number
                               of heads for upsampling. Deprecated.
    :param use_scale_shift_norm: use a FiLM-like conditioning mechanism.
    :param resblock_updown: use residual blocks for up/downsampling.
    :param use_new_attention_order: use a different attention pattern for potentially
                                    increased efficiency.
    """

    def __init__(
        self,
        image_size,
        in_channels,
        model_channels,
        out_channels,
        num_res_blocks,
        attention_resolutions,
        dropout=0,
        channel_mult=(1, 2, 4, 8),
        conv_resample=True,
        dims=2,
        num_classes=None,
        use_checkpoint=False,
        use_fp16=False,
        num_heads=1,
        num_head_channels=-1,
        num_heads_upsample=-1,
        use_scale_shift_norm=False,
        resblock_updown=False,
        use_new_attention_order=False,
    ):
        super().__init__()

        if num_heads_upsample == -1:
            num_heads_upsample = num_heads

        self.image_size = image_size
        self.in_channels = in_channels
        self.model_channels = model_channels
        self.out_channels = out_channels
        self.num_res_blocks = num_res_blocks
        self.attention_resolutions = attention_resolutions
        self.dropout = dropout
        self.channel_mult = channel_mult
        self.conv_resample = conv_resample
        self.num_classes = num_classes
        self.use_checkpoint = use_checkpoint
        self.dtype = th.float16 if use_fp16 else th.float32
        self.num_heads = num_heads
        self.num_head_channels = num_head_channels
        self.num_heads_upsample = num_heads_upsample

        time_embed_dim = model_channels * 4
        self.time_embed = nn.Sequential(
            linear(model_channels, time_embed_dim),
            nn.SiLU(),
            linear(time_embed_dim, time_embed_dim),
        )

        if self.num_classes is not None:
            self.label_emb = nn.Embedding(num_classes, time_embed_dim)

        ch = input_ch = int(channel_mult[0] * model_channels)
        self.input_blocks = nn.ModuleList(
            [TimestepEmbedSequential(conv_nd(dims, in_channels, ch, 3, padding=1))]
        )
        self._feature_size = ch
        input_block_chans = [ch]
        ds = 1
        for level, mult in enumerate(channel_mult):
            for _ in range(num_res_blocks):
                layers = [
                    ResBlock(
                        ch,
                        time_embed_dim,
                        dropout,
                        out_channels=int(mult * model_channels),
                        dims=dims,
                        use_checkpoint=use_checkpoint,
                        use_scale_shift_norm=use_scale_shift_norm,
                    )
                ]
                ch = int(mult * model_channels)
                if ds in attention_resolutions:
                    layers.append(
                        AttentionBlock(
                            ch,
                            use_checkpoint=use_checkpoint,
                            num_heads=num_heads,
                            num_head_channels=num_head_channels,
                            use_new_attention_order=use_new_attention_order,
                        )
                    )
                self.input_blocks.append(TimestepEmbedSequential(*layers))
                self._feature_size += ch
                input_block_chans.append(ch)
            if level != len(channel_mult) - 1:
                out_ch = ch
                self.input_blocks.append(
                    TimestepEmbedSequential(
                        ResBlock(
                            ch,
                            time_embed_dim,
                            dropout,
                            out_channels=out_ch,
                            dims=dims,
                            use_checkpoint=use_checkpoint,
                            use_scale_shift_norm=use_scale_shift_norm,
                            down=True,
                        )
                        if resblock_updown
                        else Downsample(
                            ch, conv_resample, dims=dims, out_channels=out_ch
                        )
                    )
                )
                ch = out_ch
                input_block_chans.append(ch)
                ds *= 2
                self._feature_size += ch

        self.middle_block = TimestepEmbedSequential(
            ResBlock(
                ch,
                time_embed_dim,
                dropout,
                dims=dims,
                use_checkpoint=use_checkpoint,
                use_scale_shift_norm=use_scale_shift_norm,
            ),
            AttentionBlock(
                ch,
                use_checkpoint=use_checkpoint,
                num_heads=num_heads,
                num_head_channels=num_head_channels,
                use_new_attention_order=use_new_attention_order,
            ),
            ResBlock(
                ch,
                time_embed_dim,
                dropout,
                dims=dims,
                use_checkpoint=use_checkpoint,
                use_scale_shift_norm=use_scale_shift_norm,
            ),
        )
        self._feature_size += ch

        self.output_blocks = nn.ModuleList([])
        for level, mult in list(enumerate(channel_mult))[::-1]:
            for i in range(num_res_blocks + 1):
                ich = input_block_chans.pop()
                layers = [
                    ResBlock(
                        ch + ich,
                        time_embed_dim,
                        dropout,
                        out_channels=int(model_channels * mult),
                        dims=dims,
                        use_checkpoint=use_checkpoint,
                        use_scale_shift_norm=use_scale_shift_norm,
                    )
                ]
                ch = int(model_channels * mult)
                if ds in attention_resolutions:
                    layers.append(
                        AttentionBlock(
                            ch,
                            use_checkpoint=use_checkpoint,
                            num_heads=num_heads_upsample,
                            num_head_channels=num_head_channels,
                            use_new_attention_order=use_new_attention_order,
                        )
                    )
                if level and i == num_res_blocks:
                    out_ch = ch
                    layers.append(
                        ResBlock(
                            ch,
                            time_embed_dim,
                            dropout,
                            out_channels=out_ch,
                            dims=dims,
                            use_checkpoint=use_checkpoint,
                            use_scale_shift_norm=use_scale_shift_norm,
                            up=True,
                        )
                        if resblock_updown
                        else Upsample(ch, conv_resample, dims=dims, out_channels=out_ch)
                    )
                    ds //= 2
                self.output_blocks.append(TimestepEmbedSequential(*layers))
                self._feature_size += ch

        self.out = nn.Sequential(
            normalization(ch),
            nn.SiLU(),
            zero_module(conv_nd(dims, input_ch, out_channels, 3, padding=1)),
        )

    def convert_to_fp16(self):
        self.input_blocks.apply(convert_module_to_f16)
        self.middle_block.apply(convert_module_to_f16)
        self.output_blocks.apply(convert_module_to_f16)

    def convert_to_fp32(self):
        self.input_blocks.apply(convert_module_to_f32)
        self.middle_block.apply(convert_module_to_f32)
        self.output_blocks.apply(convert_module_to_f32)

    def forward(self, x, timesteps, y=None):
        """
        Apply the model to an input batch.
        :param x: an [N x C x ...] Tensor of inputs.
        :param timesteps: a 1-D batch of timesteps.
        :param y: an [N] Tensor of labels, if class-conditional.
        :return: an [N x C x ...] Tensor of outputs.
        """
        assert (y is not None) == (
            self.num_classes is not None
        ), "must specify y if and only if the model is class-conditional"

        hs = []
        emb = self.time_embed(timestep_embedding(timesteps, self.model_channels))

        if self.num_classes is not None:
            assert y.shape == (x.shape[0],)
            emb = emb + self.label_emb(y)

        h = x.type(self.dtype)
        for module in self.input_blocks:
            h = module(h, emb)
            hs.append(h)
        h = self.middle_block(h, emb)
        for module in self.output_blocks:
            h = th.cat([h, hs.pop()], dim=1)
            h = module(h, emb)
        h = h.type(x.dtype)
        return self.out(h)

def create_model(
    image_size,
    num_channels,
    num_res_blocks,
    channel_mult="",
    learn_sigma=False,
    class_cond=False,
    use_checkpoint=False,
    attention_resolutions="16",
    num_heads=1,
    num_head_channels=-1,
    num_heads_upsample=-1,
    use_scale_shift_norm=False,
    dropout=0,
    resblock_updown=False,
    use_fp16=False,
    use_new_attention_order=False,
    in_channels=6,
    out_channels=3,
):
    if channel_mult == "":
        if image_size == 512:
            channel_mult = (0.5, 1, 1, 2, 2, 4, 4)
        elif image_size == 256:
            channel_mult = (1, 1, 2, 2, 4, 4)
        elif image_size == 128:
            channel_mult = (1, 1, 2, 3, 4)
        elif image_size == 64:
            channel_mult = (1, 2, 3, 4)
        else:
            raise ValueError(f"unsupported image size: {image_size}")

    attention_ds = []
    for res in attention_resolutions.split(","):

    return UNetModel(
        image_size=image_size,
        in_channels=in_channels,
        model_channels=num_channels,

        num_res_blocks=num_res_blocks,

        channel_mult=channel_mult,
        num_classes=(NUM_CLASSES if class_cond else None),

    )

### D 2. Forward diffusion và reverse diffusion
Ở training, chọn một timestep`t` rồi thêm nhiễu:

$$x_t=\sqrt{\bar\alpha_t}x_0+\sqrt{1-\bar\alpha_t}\epsilon.$$

U-Net học dự đoán epsilon; loss gốc là meanL 1. Sampling bắt đầu từ Gaussian noise và chạy ngược 250 bước. Một optimizer update không tương đương một lần sinh ảnh 250 bước.

def cosine_beta_schedule(timesteps, s=0.008):
    steps = timesteps + 1
    x = np.linspace(0, steps, steps)
    alphas_cumprod = np.cos(((x / steps) + s) / (1 + s) * np.pi * 0.5) ** 2
    alphas_cumprod = alphas_cumprod / alphas_cumprod[0]
    betas = 1 - (alphas_cumprod[1:] / alphas_cumprod[:-1])
    return np.clip(betas, a_min=0, a_max=0.999)

class GaussianDiffusion(nn.Module):
    def __init__(
        self,
        denoise_fn,
        *,
        image_size,
        channels=3,
        timesteps=1000,
        loss_type="l1",
        betas=None,
    ):
        super().__init__()
        self.channels = channels
        self.image_size = image_size
        self.denoise_fn = denoise_fn

        if exists(betas):
            betas = (
                betas.detach().cpu().numpy()
                if isinstance(betas, torch.Tensor)
                else betas
            )
        else:
            betas = cosine_beta_schedule(timesteps)

        alphas = 1.0 - betas
        alphas_cumprod = np.cumprod(alphas, axis=0)
        alphas_cumprod_prev = np.append(1.0, alphas_cumprod[:-1])

        (timesteps,) = betas.shape
        self.num_timesteps = int(timesteps)
        self.loss_type = loss_type

        to_torch = partial(torch.tensor, dtype=torch.float32)

        self.register_buffer("betas", to_torch(betas))
        self.register_buffer("alphas_cumprod", to_torch(alphas_cumprod))
        self.register_buffer("alphas_cumprod_prev", to_torch(alphas_cumprod_prev))

        # calculations for diffusion q(x_t | x_{t-1}) and others
        self.register_buffer("sqrt_alphas_cumprod", to_torch(np.sqrt(alphas_cumprod)))
        self.register_buffer(
            "sqrt_one_minus_alphas_cumprod", to_torch(np.sqrt(1.0 - alphas_cumprod))
        )
        self.register_buffer(
            "log_one_minus_alphas_cumprod", to_torch(np.log(1.0 - alphas_cumprod))
        )
        self.register_buffer(
            "sqrt_recip_alphas_cumprod", to_torch(np.sqrt(1.0 / alphas_cumprod))
        )
        self.register_buffer(
            "sqrt_recipm1_alphas_cumprod", to_torch(np.sqrt(1.0 / alphas_cumprod - 1))
        )

        # calculations for posterior q(x_{t-1} | x_t, x_0)
        posterior_variance = (
            betas * (1.0 - alphas_cumprod_prev) / (1.0 - alphas_cumprod)
        )
        # above: equal to 1. / (1. / (1. - alpha_cumprod_tm1) + alpha_t / beta_t)
        self.register_buffer("posterior_variance", to_torch(posterior_variance))
        # below: log calculation clipped because the posterior variance is 0 at the beginning of the diffusion chain
        self.register_buffer(
            "posterior_log_variance_clipped",
            to_torch(np.log(np.maximum(posterior_variance, 1e-20))),
        )
        self.register_buffer(
            "posterior_mean_coef1",
            to_torch(betas * np.sqrt(alphas_cumprod_prev) / (1.0 - alphas_cumprod)),
        )
        self.register_buffer(
            "posterior_mean_coef2",
            to_torch(
                (1.0 - alphas_cumprod_prev) * np.sqrt(alphas) / (1.0 - alphas_cumprod)
            ),
        )

    def q_mean_variance(self, x_start, t):
        mean = extract(self.sqrt_alphas_cumprod, t, x_start.shape) * x_start
        variance = extract(1.0 - self.alphas_cumprod, t, x_start.shape)
        log_variance = extract(self.log_one_minus_alphas_cumprod, t, x_start.shape)
        return mean, variance, log_variance

    def predict_start_from_noise(self, x_t, t, noise):
        return (
            extract(self.sqrt_recip_alphas_cumprod, t, x_t.shape) * x_t
            - extract(self.sqrt_recipm1_alphas_cumprod, t, x_t.shape) * noise
        )

    def q_posterior(self, x_start, x_t, t, c=None):
        posterior_mean = (
            extract(self.posterior_mean_coef1, t, x_t.shape) * x_start
            + extract(self.posterior_mean_coef2, t, x_t.shape) * x_t
        )
        posterior_variance = extract(self.posterior_variance, t, x_t.shape)
        posterior_log_variance_clipped = extract(
            self.posterior_log_variance_clipped, t, x_t.shape
        )
        return posterior_mean, posterior_variance, posterior_log_variance_clipped

    def p_mean_variance(self, x, t, clip_denoised: bool, c=None):
        x_recon = self.predict_start_from_noise(
            x, t=t, noise=self.denoise_fn(torch.cat([x, c], 1), t)
        )

        if clip_denoised:
            x_recon.clamp_(-1.0, 1.0)

        model_mean, posterior_variance, posterior_log_variance = self.q_posterior(
            x_start=x_recon, x_t=x, t=t, c=c
        )
        return model_mean, posterior_variance, posterior_log_variance

    @torch.no_grad()
    def p_sample(
        self, x, t, condition_tensors=None, clip_denoised=True, repeat_noise=False
    ):
        b, *_, device = *x.shape, x.device
        model_mean, _, model_log_variance = self.p_mean_variance(
            x=x, t=t, c=condition_tensors, clip_denoised=clip_denoised
        )
        noise = noise_like(x.shape, device, repeat_noise)
        # no noise when t == 0
        nonzero_mask = (1 - (t == 0).float()).reshape(b, *((1,) * (len(x.shape) - 1)))
        return model_mean + nonzero_mask * (0.5 * model_log_variance).exp() * noise

    @torch.no_grad()
    def p_sample_loop(self, shape, condition_tensors=None):
        device = self.betas.device
        b = shape[0]
        img = torch.randn(shape, device=device)

        # Looping through timesteps in reverse order
        for i in tqdm(
            reversed(range(self.num_timesteps)),
            desc="sampling loop time step",
            total=self.num_timesteps,
        ):
            t = torch.full((b,), i, device=device, dtype=torch.long)
            img = self.p_sample(img, t, condition_tensors=condition_tensors)

        return img

    @torch.no_grad()
    def sample(self, batch_size=16, condition_tensors=None):
        image_size = self.image_size
        channels = self.channels

        return self.p_sample_loop(
            (batch_size, channels, image_size, image_size),
            condition_tensors=condition_tensors,
        )

    @torch.no_grad()
    def interpolate(self, x1, x2, t=None, lam=0.5):
        b, *_, device = *x1.shape, x1.device
        t = default(t, self.num_timesteps - 1)

        assert x1.shape == x2.shape

        t_batched = torch.stack([torch.tensor(t, device=device)] * b)
        xt1, xt2 = map(lambda x: self.q_sample(x, t=t_batched), (x1, x2))

        img = (1 - lam) * xt1 + lam * xt2
        for i in tqdm(
            reversed(range(0, t)), desc="interpolation sample time step", total=t
        ):
            img = self.p_sample(
                img, torch.full((b,), i, device=device, dtype=torch.long)
            )

        return img

    def q_sample(self, x_start, t, noise=None):
        noise = default(noise, lambda: torch.randn_like(x_start))

        return (
            extract(self.sqrt_alphas_cumprod, t, x_start.shape) * x_start
            + extract(self.sqrt_one_minus_alphas_cumprod, t, x_start.shape) * noise
        )

    def p_losses(self, x_start, t, condition_tensors=None, noise=None):
        b, c, h, w = x_start.shape
        noise = default(noise, lambda: torch.randn_like(x_start))

        x_noisy = self.q_sample(x_start=x_start, t=t, noise=noise)
        x_recon = self.denoise_fn(torch.cat([x_noisy, condition_tensors], 1), t)

        if self.loss_type == "l1":
            loss = (noise - x_recon).abs().mean()
        elif self.loss_type == "l2":
            loss = F.mse_loss(x_recon, noise)
        else:
            raise NotImplementedError()

        return loss

    def forward(self, x, condition_tensors=None, *args, **kwargs):
        (
            b,
            c,
            h,
            w,
            device,
            img_size,
        ) = (
            *x.shape,
            x.device,
            self.image_size,
        )
        assert (
            h == img_size and w == img_size
        ), f"height and width of image must be {img_size}"
        t = torch.randint(0, self.num_timesteps, (b,), device=device).long()
        return self.p_losses(x, t, condition_tensors=condition_tensors, *args, **kwargs)

@torch.no_grad()
def full_reverse(diff, condition, seed, trace=False):
    """One full 250-step trajectory with its own RNG scope; [0,1] RGB out."""
    diff.eval()

def train_own(cfg):
    rank, world, device = runtime(cfg, True)
    records = {r["id"]: r for r in cfg["records"]}
    diff = factory(cfg, device)
    ema = copy.deepcopy(diff).eval().requires_grad_(False)
    net = (
        DDP(
            diff.denoise_fn,
            device_ids=[device.index] if device.type == "cuda" else None,
            broadcast_buffers=False,
        )
        if world > 1
        else diff.denoise_fn
    )
    # DDP broadcasts the randomly initialized network from rank0; EMA must match it.
    ema.load_state_dict(diff.state_dict())
    opt = torch.optim.Adam(net.parameters(), lr=cfg["own_lr"])
    path = Path(cfg["out"]) / "own/checkpoints/last.pt"
    step = 0
    history = []
    if path.exists():
        cp = load_pt(path)
        if cp["protocol"] != cfg["protocol"] or cp["world_size"] != world:
            raise ValueError("Wrong own-training resume")
        diff.load_state_dict(cp["model"])
        ema.load_state_dict(cp["ema"])
        opt.load_state_dict(cp["optimizer"])
        step, history = cp["step"], cp["history"]
        restore_rng(cp["rng_by_rank"][rank], device)
        del cp
    loaded = step
    if not path.exists():
        checkpoint_own(cfg, diff, ema, opt, step, history, rank, world, device)
    batch = cfg["own_per_gpu_batch"]
    acc = cfg["own_accumulation"]
    while step < cfg["own_steps"]:
        if resource_stop(cfg, device, True):
            break
        tick = time.monotonic()
        net.train()
        opt.zero_grad(set_to_none=True)
        loss_value = 0.0
        for micro in range(acc):
            positions = [
                step * batch * world * acc + micro * batch * world + rank * batch + j
                for j in range(batch)
            ]
            ids = stream_ids(cfg["train_ids"], positions, cfg["seed"])
            rows = [
                tensor_pair(records[sid], cfg["size"], cfg["seed"] + pos)
                for sid, pos in zip(ids, positions)
            ]
            clean = torch.stack([x for x, y in rows]).to(device) * 2 - 1
            condition = (
                torch.stack([y.repeat(3, 1, 1) for x, y in rows]).to(device) * 2 - 1
            )
            scope = (
                net.no_sync()
                if world > 1 and micro < acc - 1
                else contextlib.nullcontext()
            )
            with scope:
                t = torch.randint(0, 250, (len(clean),), device=device)
                noise = torch.randn_like(clean)
                noisy = diff.q_sample(clean, t, noise)
                eps = net(torch.cat([noisy, condition], 1), t)
                loss = (eps - noise).abs().mean()
                invalid = torch.tensor(int(not torch.isfinite(loss)), device=device)
                if world > 1:
                    dist.all_reduce(invalid)
                if invalid.item():
                    raise RuntimeError("Nonfinite DDPM loss")
                (loss / acc).backward()
                loss_value += float(loss.detach()) / acc
            del clean, condition, noise, noisy, eps, loss, t
        norm = torch.nn.utils.clip_grad_norm_(
            net.parameters(), 1.0, error_if_nonfinite=True
        )
        opt.step()
        with torch.no_grad():
            for a, b in zip(ema.parameters(), diff.parameters()):
                a.mul_(0.995).add_(b, alpha=0.005)
            for a, b in zip(ema.buffers(), diff.buffers()):
                a.copy_(b)
        if world > 1:
            value = torch.tensor(loss_value, device=device)
            dist.all_reduce(value)
            loss_value = float(value / world)
        step += 1
        history.append(
            {
                "step": step,
                "loss": loss_value,
                "grad_norm": float(norm),
                "seconds": time.monotonic() - tick,
            }
        )
        if step % 25 == 0:
            print(
                f'OWN rank{rank}: {step}/{cfg["own_steps"]}, loss={loss_value:.5f}',
                flush=True,
            )
        if step % cfg["save_every"] == 0 or step == cfg["own_steps"]:
            checkpoint_own(cfg, diff, ema, opt, step, history, rank, world, device)
    if step != loaded:
        checkpoint_own(cfg, diff, ema, opt, step, history, rank, world, device)
    if rank == 0:
        write_json(
            Path(cfg["out"]) / "own/status.json",
            {
                "step": step,
                "complete": step == cfg["own_steps"],
                "protocol": cfg["protocol"],
                "additional_updates_this_launch": step - loaded,
            },
        )
    telemetry(
        cfg,
        "own_train",
        rank,
        device,
        step=step,
        full_state_sha256=state_fingerprint(diff.state_dict()),
    )
    if dist.is_initialized():
        dist.destroy_process_group()

def palette_projection(x, palette, gain_limit=3.0):
    support = tissue_support(x)
    source = lab_moments(x, support)
    lab = rgb_to_lab(x)
    floor = x.new_tensor([0.035, 0.015, 0.015])[None]
    src_std = torch.maximum(source[:, 3:], floor)
    target_std = torch.maximum(palette[:, 3:], floor)
    scale = (target_std / src_std).clamp(1 / gain_limit, gain_limit)
    corrected = (lab - source[:, :3, None, None]) * scale[:, :, None, None] + palette[
        :, :3, None, None
    ]
    # Preserve deep shadows continuously; no threshold-based hole cutting/FOV segmentation.
    lightness = torch.minimum(corrected[:, :1], lab[:, :1].clamp_min(0) * 3.0)
    corrected = torch.cat([lightness, corrected[:, 1:]], dim=1)
    raw = lab_to_rgb(corrected, clip=False)
    result = raw.clamp(0, 1)
    # Keep existing black borders and pure white highlights at the same locations.
    preserve = (x.amax(1, keepdim=True) <= 0.025) | (x.amin(1, keepdim=True) >= 0.995)
    result = torch.where(preserve, x, result)
    penalty_mask = (~preserve).float()
    excess = ((F.relu(-raw) + F.relu(raw - 1)) * penalty_mask).sum() / (
        penalty_mask.sum() * 3
    ).clamp_min(1)
    return result, excess

class PaletteLUT(nn.Module):
    def __init__(self, dim=17, bases=3):
        super().__init__()
        self.dim, self.bases = dim, bases
        self.encoder = nn.Sequential(
            nn.Conv2d(3, 16, 3, 2, 1),
            nn.SiLU(),
            nn.Conv2d(16, 32, 3, 2, 1),
            nn.SiLU(),
            nn.Conv2d(32, 48, 3, 2, 1),
            nn.SiLU(),
            nn.AdaptiveAvgPool2d(1),
        )
        self.predictor = nn.Sequential(
            nn.Linear(48 + 6, 48), nn.SiLU(), nn.Linear(48, bases)
        )
        nn.init.zeros_(self.predictor[-1].weight)
        nn.init.zeros_(self.predictor[-1].bias)
        values = torch.randn(bases, 3, dim, dim, dim) * 0.002
        values -= values.mean(0, keepdim=True)
        self.residuals = nn.Parameter(values)
        self.register_buffer("identity", identity_lut(dim))

    def forward(self, x, palette, project=True):
        features = self.encoder(
            F.interpolate(x, size=(64, 64), mode="bilinear", align_corners=False)
        ).flatten(1)
        weights = self.predictor(torch.cat([features, palette], 1)).softmax(1)
        tables = self.identity[None] + 0.75 * self.residuals.tanh()
        lut = torch.einsum("bm,mcijk->bcijk", weights, tables)
        raw = apply_lut(x, lut)
        clip_penalty = (F.relu(-raw) + F.relu(raw - 1)).mean()
        rendered = raw.clamp(0, 1)
        if project:
            rendered, gamut = palette_projection(rendered, palette)
            clip_penalty = clip_penalty + gamut
            baseline, _ = palette_projection(x, palette)
            # Learned increment is bounded to avoid large new chromatic speckles.
            rendered = baseline + 0.04 * torch.tanh((rendered - baseline) / 0.04)
            rendered = rendered.clamp(0, 1)
            preserve = (x.amax(1, keepdim=True) <= 0.025) | (
                x.amin(1, keepdim=True) >= 0.995
            )
            rendered = torch.where(preserve, x, rendered)
        return rendered, clip_penalty

    def regularization(self):
        tables = self.identity[None] + 0.75 * self.residuals.tanh()
        tv = sum((self.residuals.diff(dim=axis) ** 2).mean() for axis in [2, 3, 4])
        # Monotonicity in the corresponding color axis; cross-channel mixing is allowed.
        mono = F.relu(-tables[:, 0].diff(dim=3)).mean()
        mono += F.relu(-tables[:, 1].diff(dim=2)).mean()
        mono += F.relu(-tables[:, 2].diff(dim=1)).mean()
        return tv, mono

Paired reconstruction giữ nội dung trên cặp ảnh thật–ảnh bị biến đổi màu; SWD điều chỉnh phân bố palette; gradient và regularization hạn chế thay cấu trúc. Teacher chỉ cung cấp tham chiếu màu trong nhánh calibration, không chép hình học vào output.

def color_objective(corrector, noisy, clean, style, raw, teacher, teacher_style):
    repaired, gamut1 = corrector(noisy, style)
    corrected, gamut2 = corrector(raw, teacher_style)
    paired = (
        (repaired - clean).abs().mean()
        + 0.2 * (rgb_to_lab(repaired) - rgb_to_lab(clean)).abs().mean()
        + 0.1 * gradient_match(repaired, clean)
    )
    distribution = palette_swd(corrected, teacher)
    structure = gradient_match(normalized_luma(corrected), normalized_luma(raw))
    analytic, _ = palette_projection(raw, teacher_style)
    chroma_structure = gradient_match(rgb_to_lab(corrected), rgb_to_lab(analytic))
    tv, mono = corrector.regularization()
    loss = (
        paired
        + 0.5 * distribution
        + 0.01 * structure
        + 0.05 * chroma_structure
        + 0.01 * (gamut1 + gamut2)
        + 1e-3 * tv
        + 1e-3 * mono
    )
    return loss, {"paired": paired, "palette_swd": distribution, "structure": structure}

def degrade_real(clean, condition, seed):
    """
    Paired blur+grain simulation, especially background. No geometry warp or color shift.
    This small original degradation model is inspired by restoration training, not BSRGAN reproduction.
    """
    gen = torch.Generator(device=clean.device).manual_seed(int(seed))
    result = []
    rgb = (clean + 1) / 2
    mask = ((condition[:, :1] + 1) / 2).clamp(0, 1)
    for i in range(len(clean)):
        sigma = 0.5 + 1.0 * float(torch.rand((), device=clean.device, generator=gen))
        axis = torch.arange(-3, 4, device=clean.device, dtype=clean.dtype)
        kernel = torch.exp(-axis.square() / (2 * sigma * sigma))
        kernel /= kernel.sum()
        k = (kernel[:, None] * kernel[None, :])[None, None].repeat(3, 1, 1, 1)
        blur = F.conv2d(
            F.pad(rgb[i : i + 1], (3, 3, 3, 3), mode="reflect"), k, groups=3
        )
        gaussian = torch.randn(rgb[i : i + 1].shape, device=clean.device, generator=gen)
        grain = (
            0.007 + 0.018 * float(torch.rand((), device=clean.device, generator=gen))
        ) * gaussian
        speckle = (
            0.025
            * torch.randn((1, 1, *rgb.shape[-2:]), device=clean.device, generator=gen)
            * rgb[i : i + 1]
        )
        mix = (1 - 0.65 * mask[i : i + 1]) * tissue_support(rgb[i : i + 1])
        # Preserve genuine black pixels; no claim that simulated artifacts exactly match DDPM outputs.
        degraded = rgb[i : i + 1] + mix * (blur + grain + speckle - rgb[i : i + 1])
        degraded = torch.where(
            rgb[i : i + 1].amax(1, keepdim=True) < 0.025, rgb[i : i + 1], degraded
        )
        result.append(degraded.clamp(0, 1))
    return torch.cat(result) * 2 - 1


def mixed_training_state(diff, clean, condition, t, cfg, seed):
    gen = torch.Generator(device=clean.device).manual_seed(int(seed))
    repair = (
        torch.rand((len(clean), 1, 1, 1), device=clean.device, generator=gen)
        < cfg["restoration_probability"]
    )
    repair = repair & (t >= cfg["restoration_t_min"])[:, None, None, None]
    corrupted = degrade_real(clean, condition, seed + 1)
    base = torch.where(repair, corrupted, clean)
    noise = torch.randn(clean.shape, device=clean.device, generator=gen)
    noisy = diff.q_sample(base, t, noise)
    # A corrupted-real branch must NOT incorrectly use plain Gaussian noise as target.
    alpha = extract(diff.sqrt_alphas_cumprod, t, clean.shape)
    sigma = extract(diff.sqrt_one_minus_alphas_cumprod, t, clean.shape).clamp_min(1e-5)
    effective_target = (noisy - alpha * clean) / sigma
    return noisy, effective_target, repair.flatten()

def protection_regions(condition, cfg):
    mask = ((condition[:, :1] + 1) / 2).clamp(0, 1)
    radius = cfg["guard_radius"]
    dilated = F.max_pool2d(mask, 2 * radius + 1, 1, radius)
    erode_radius = max(1, radius // 2)
    eroded = -F.max_pool2d(-mask, 2 * erode_radius + 1, 1, erode_radius)
    band = (dilated - eroded).clamp(0, 1)
    soft = F.avg_pool2d(dilated, 9, 1, 4)
    return mask, band, soft

def continuation_loss(
    diff, eps, noisy, clean, target, t, condition, per, cfg, teacher_eps=None, pure=None
):
    # Teacher is the frozen selected v2 refiner, not a new author initialization.
    total, parts = v2_background_loss(
        diff, eps, noisy, clean, target, t, condition, per, cfg, None, None
    )
    mask, band, _ = protection_regions(condition, cfg)
    valid = tissue_support((clean + 1) / 2).detach()
    band = band * valid
    prior = eps.sum() * 0
    if teacher_eps is not None and pure is not None and pure.any():
        weights = 1 + cfg["prior_fg_boost"] * mask + cfg["prior_band_boost"] * band
        prior = masked_mean((eps[pure] - teacher_eps[pure]).abs(), weights[pure]).mean()
    pred = diff.predict_start_from_noise(noisy, t, eps).clamp(-1, 1)
    eligible = t <= cfg["aux_t_max"]
    boundary_rgb = eps.sum() * 0
    boundary_grad = boundary_rgb
    if eligible.any():
        a, b, w = pred[eligible], clean[eligible], band[eligible]
        boundary_rgb = masked_mean((a - b).abs(), w).mean()
        al = rgb_to_lab((a + 1) / 2)[:, :1]
        bl = rgb_to_lab((b + 1) / 2)[:, :1]
        boundary_grad = region_gradient(al, bl, w).mean()
    total = (
        total
        + cfg["teacher_weight"] * prior
        + cfg["boundary_rgb_weight"] * boundary_rgb
        + cfg["boundary_gradient_weight"] * boundary_grad
    )
    parts = torch.cat(
        [parts[:3], prior[None], parts[4:], boundary_rgb[None], boundary_grad[None]]
    )
    return total, parts

@torch.no_grad()
def protected_reverse(diff, reference, condition, steps, seed, cfg, palette=None):
    # Protect only coarse structure near the polyp; do not copy real-reference pixels.
    if not 1 <= steps <= diff.num_timesteps:
        raise ValueError("Invalid reverse steps")
    clean = reference * 2 - 1
    gen = torch.Generator(device=reference.device).manual_seed(int(seed))
    noise = torch.randn(clean.shape, device=clean.device, generator=gen)
    ts = torch.full((len(clean),), steps - 1, device=clean.device, dtype=torch.long)
    x = diff.q_sample(clean, ts, noise)
    mask, band, protected = protection_regions(condition, cfg)
    side = 32 if reference.shape[-1] >= 32 else 4
    for t in reversed(range(steps)):
        ts = torch.full((len(x),), t, device=x.device, dtype=torch.long)
        eps = diff.denoise_fn(torch.cat([x, condition], 1), ts)
        x0 = diff.predict_start_from_noise(x, ts, eps).clamp(-1, 1)
        mean, _, logvar = diff.q_posterior(x0, x, ts)
        z = torch.randn(x.shape, device=x.device, generator=gen)
        x = mean + (t > 0) * (0.5 * logvar).exp() * z
        fade = max(
            0.0,
            (t - cfg["release_last_steps"])
            / max(steps - 1 - cfg["release_last_steps"], 1),
        )
        alpha = diff.alphas_cumprod[t - 1] if t else x.new_tensor(1.0)
        ref = alpha.sqrt() * clean + (1 - alpha).sqrt() * noise
        coarse_delta = low_pass(ref, side) - low_pass(x, side)
        bg_delta = low_pass(ref, cfg["low_side"]) - low_pass(x, cfg["low_side"])
        fg_strength = (
            fade * cfg["foreground_anchor"] + (1 - fade) * cfg["foreground_floor"]
        )
        guard_strength = (1 - fade) * cfg["boundary_floor"]
        weight = (fg_strength * mask + guard_strength * band).clamp(0, 1)
        # Background remains fully released in the last detail steps, as in v2.
        x = (
            x
            + weight * coarse_delta
            + fade * cfg["background_anchor"] * (1 - protected) * bg_delta
        )
    out = ((x + 1) / 2).clamp(0, 1)
    return palette_chroma_only(
        out, reference if palette is None else palette, cfg["chroma_strength"]
    )

def checkpoint(cfg, diff, opt, step, history, best, rank, world, device):
    local = state_rng(device)
    states = [None] * world
    if world > 1:

            "step": step,

            "initial_refiner_updates": cfg["initial_refiner_updates"],

        # Keep only baseline and latest committed BEST backup for crash rollback.
        committed = max([r["step"] for r in history if r.get("improved")] or [0])
        for previous in (Path(cfg["out"]) / "checkpoints").glob("best_*.pt"):

            save_pt(Path(cfg["out"]) / "checkpoints" / f"step_{step}.pt", value)
    if world > 1:
        dist.barrier()

def refiner_update(
    cfg, diff, teacher, per, net, opt, records, step, rank, world, device
):
    tick = time.monotonic()
    diff.train()
    # A new objective starts a new optimizer; same-study resume restores it exactly.
    phase = min(1.0, step / max(cfg["train_steps"] - 1, 1))
    lr = cfg["lr_min"] + 0.5 * (cfg["lr"] - cfg["lr_min"]) * (
        1 + math.cos(math.pi * phase)
    )
    warm = min(1.0, (step + 1) / max(cfg["warmup_updates"], 1))
    for group in opt.param_groups:
        group["lr"] = lr * warm
    opt.zero_grad(set_to_none=True)
    components = torch.zeros(7, device=device)
    seen = []
    for micro in range(cfg["accumulation"]):
        sync = (
            contextlib.nullcontext()
            if world == 1 or micro == cfg["accumulation"] - 1
            else net.no_sync()
        )
        with sync:
            clean, condition, ids = training_batch(
                cfg, records, step, rank, world, micro
            )
            seen += ids
            clean, condition = clean.to(device), condition.to(device)
            t = torch.randint(0, cfg["train_t_max"] + 1, (len(clean),), device=device)
            noisy, target, repair = mixed_training_state(
                diff,
                clean,
                condition,
                t,
                cfg,
                cfg["seed"] + step * 10000 + rank * 100 + micro,
            )
            with torch.no_grad():
                teacher_eps = teacher.denoise_fn(torch.cat([noisy, condition], 1), t)
            eps = net(torch.cat([noisy, condition], 1), t)
            loss, parts = continuation_loss(
                diff,
                eps,
                noisy,
                clean,
                target,
                t,
                condition,
                per,
                cfg,
                teacher_eps,
                ~repair,
            )
            finite = all_reduce(
                torch.tensor(int(not torch.isfinite(loss)), device=device)
            )
            if finite.item():
                raise RuntimeError("Nonfinite training loss on at least one rank")
            (loss / cfg["accumulation"]).backward()
            components += parts.detach() / cfg["accumulation"]
            del (
                clean,
                condition,
                t,
                target,
                repair,
                noisy,
                teacher_eps,
                eps,
                loss,
                parts,
            )
    grad = torch.nn.utils.clip_grad_norm_(
        net.parameters(), 1.0, error_if_nonfinite=True
    )
    opt.step()
    step += 1
    all_reduce(components)
    components /= world
    seconds = (
        all_reduce(torch.tensor(time.monotonic() - tick, device=device)).item() / world
    )
    row = {
        "step": step,
        "epsilon_L1": float(components[0]),
        "gradient_L1": float(components[1]),
        "LPIPS": float(components[2]),
        "prior_L1": float(components[3]),
        "restoration_RGB_L1": float(components[4]),
        "boundary_RGB_L1": float(components[5]),
        "boundary_gradient_L1": float(components[6]),
        "lr": opt.param_groups[0]["lr"],
        "seconds_update": seconds,
        "grad_norm": float(grad),
    }
    return row, len(seen)

    path = Path(cfg["out"]) / "checkpoints/last.pt"
    step = 0
    history = []
    best = float("inf")
    if path.exists():
        cp = load_pt(path)

            cp["protocol"]
            not in [cfg["protocol"], *cfg.get("accepted_resume_protocols", [])]
            or cp["world_size"] != world
        ):
            raise ValueError("Checkpoint/protocol/world size mismatch")
        diff.load_state_dict(cp["model"])
        opt.load_state_dict(cp["optimizer"])

            selected_step = max(

def kernel_mmd(a, b):
    a, b = np.asarray(a, float), np.asarray(b, float)
    d = ((b[:, None] - b[None, :]) ** 2).mean(-1)
    positive = d[d > 1e-12]
    bandwidth = float(np.median(positive)) if len(positive) else 1.0

    def kernel(x, y):
        z = ((x[:, None] - y[None, :]) ** 2).mean(-1)
        return sum(np.exp(-z / (2 * bandwidth * s)) for s in [0.5, 1.0, 2.0]) / 3

    return max(
        0.0, float(kernel(a, a).mean() + kernel(b, b).mean() - 2 * kernel(a, b).mean())
    )
